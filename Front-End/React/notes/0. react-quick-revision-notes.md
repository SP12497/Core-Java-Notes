# 11. `useEffect`

```js
useEffect(cb, [params])
```

* Used to synchronize with external systems / perform side effects.
* Dependency array controls when the effect re-runs.
* Cleanup runs before the effect is re-run and when the component unmounts.

> Don't use an effect when a value can simply be calculated during rendering.

Example dependency behavior:

```js
useEffect(() => {
  // effect
}, []);
```

* Runs after the initial render; in development Strict Mode, React may run setup/cleanup an extra time.

```js
useEffect(() => {
  // effect
}, [inputValue]);
```

* Re-runs when `inputValue` changes.

**Important correction:** `setUsername` should generally **not** be placed in the dependency array just because it updates `username`. State setter functions have stable identity.

## 2. `useEffect()`

* Used for side effects such as subscriptions, timers, DOM APIs, and synchronization with external systems.
* Runs after rendering/commit.

```jsx
useEffect(() => {
	// 1. Side-effect code goes here

	return () => {
		// 2. Optional cleanup code goes here. execute before component destroys
	};
}, [dependencies]); // 3. Optional dependency array
```

### Timer Example

```jsx
const [seconds, setSeconds] = useState(0)
useEffect(() => {
	const interval = setInterval(() => {
		setSeconds((prevSeconds) => prevSeconds + 1);
	}, 1000)

	return () => {
		clearInterval(interval);
		console.log("Interval Cleared)
	}
}, [])
<p> Seconds elapsed: {seconds} </p>
```

### Parent Component

```jsx
const [toggle, setToggle] = useState(true)
{toggle && <HookExample />}
<button onClick={() => setToggle(!toggle)} Toggle </button>
```

### Output

```text
toggle is true:
```

**Quick:** Effect → setup → cleanup.

## 4. `useRef`

```jsx
	ref={inputRef}
}
```


## 6. `useReducer()`

Useful when state has multiple related transitions.

```jsx
		case "INCREMENT":

		case "DECREMENT":
			return state - 1;
		case "RESET":
			return 0;
		default:
			return state;			
	}
}

const counter = () => {
	const [count, dispatch] = useReducer(countReducer, initialState);	// return 2 fields, state and dispatch method to manage state

	button onClick={() => dispatch({ type: "INCREMENT", value: 5 })} > + </button>
	button onClick={() => dispatch({ type: "DECREMENT" })} > - </button>
	button onClick={() => dispatch({ type: "RESET" })} > Reset </button>
}
```

**Quick:** `state + action → reducer → new state`.

## 12. `useDeferredValue(value)`

* Creates a deferred version of a value.
* Useful when expensive UI can lag behind an urgent value such as typing.
* Unlike `useTransition`, it does not directly mark a state update.

```jsx

const defferedValue = useDefferedValue(input);

// Shows new results only when calculation completed.

```

**Quick:**

* `useTransition()` → defer an update.
* `useDeferredValue()` → defer a value.

## 13. `useActionState()`

* React 19 hook for managing state around an Action.
* Returns:

```text
[state, action, isPending]
```

* Useful for form/action workflows.

```jsx
const [user, submitAction, isPending] = useActionState(login, { data: null, error: null} );	// do esnt require any useState
async function login(prevState, formData) {
	const username = formData.get('username');
	const password = formData.get("password");
	try {
		const response = await loginUser(username, password);
		return { data: response.data, error: null}
	} catch (error) {
		return {...prevState, error: error.error}
	}
}

const handleSubmit = async (formData) => {	// we have to use many states to manage forms, like error, pending, state.
	setIsPending(true);
	setUser(null)
	setError(null)
	const username = formData.get('username');
	const password = formData.get("password");
	try {
		const response = await loginUser(username, password);
		setUser(response.data)
	} catch (error) {
		setError(error.error)
	}
	finally {
		setIsPending(false)
	}
}

// <form action={handleSubmit}>
<form action={submitAction}>	// action attribute is needed
	<input name="username" required />
	<input name="password" required />
	<button type="submit" disable={isPending}> {isPending? "Logging in..." : "Login" } </button>
	{user?.data &&  <p> Logged in: {user.email}} </p>
```

## 15. `useOptimistic()`

* Shows an optimistic UI update before the server confirms the operation.
* Useful when the expected server operation is asynchronous.

```jsx
const [todos, setTodos] = useState([]);
const [optimisticTodos, setOptimisticTodo] = useOptimistic( todos , 
	(oldTodos, newTodos) => [...oldTodos, {text: newTodo, pending: true}]
)

const handleAddTodo = async (formData) => {
	const newTodo = formData.get("todo");
	setOptimisticTodo(newTodo);
	await new Promise((resolve) => setTimeout(resolve, 1000));

	setTodos((currentTodos) => [
		...currentTodos,
		{text: newTodo, pending: false}
	])
}

<form>
	<input name="todo" placeholder="Enter todo..." />
	<button type="submit"> Add Todo </button>
<form>
<ul>
	{optimisticTodos.map((todo, index) => (
		<li key={index}> {todo.text} {todo.pending && <span> (Adding...) </span>}	</li>
	))}
<ul>
```

**Quick:** Show expected result immediately → reconcile with real server result.

## 16. Custom Hook

* A custom hook extracts reusable stateful logic.
* Custom hooks share **logic**, not the same state instance.

### `src/hooks/use-fetch.jsx`

```jsx
const useFetch = (cb) => {
	const [data, setData] = useState(undefined);
	const [loading, setLoading] = useState(null);
	const [error, setError] = useState(null);

	const fn = async (...args) -> {
		setLoading(true);
		setError(null) 
		try {
			const response = await cb(...args);
			setData(response)
			setError(null)
		} catch (error) {
			setError(error)
			window\.alert(error.message)
		} finally {
			setLoading(false)
		}
	};

	return {loading, error, data, fn} ;
}	
```

### `component.jsx`

```jsx
const fetchPosts = () => 
	fetch("https://jsonplaceholder.typicode.com/posts)
		.then((res) => res.json()
	);

const {data, error, loading, fn} = useFetch(fetchPosts);

useEffect(()=> {
	fn()
}, [])
```

**Quick:** Custom Hook = reusable hook-based logic.

# 17. Context API

## Problem: Prop Drilling

When data must travel through components that don't need it:

```text
Main
 ↓
ComponentA
 ↓
ComponentB
 ↓
ComponentC
```

Example:

```jsx
Component Main:
	return <ComponentA title="Sagar"/>
ComponentA
	function ComponentA({title})
		return <ComponentA title="Sagar"/>
ComponentB
	function ComponentB({title})
		return <Componentb title="Sagar"/>
ComponentC
	function ComponentC({title})
		return <div title="Sagar">
```

**Quick:** Context avoids unnecessary prop passing through intermediate components.

2. `initialState`

Reads values from Redux state.

### Step 2: Slice

`reduxToolkitTodo/src/features/todo/todoSlice.js`

```jsx
import {createSlice, nanoid } from '@reduxjs/toolkit';

const initialState = { todos: [{id: 1, text: "Hello world"}] }
export const todoSlice = createSlice(
	{
	name: 'todo',
	initialState,
	reducers: {
		addTodo: (state, action) => {		// slice methods
			const todo = {
				id: nanoid(), 
				text: action.payload
			}
			state.todos.push(todo)
		},
		removeTodo: (state, action) => {
			state.todos = state.todos.filter((todo) => todo.id !== action.payload )
		},
	}
}
//, {another feature reducer}
)

export const {addTodo, removeTodo} = todoSlice.actions

export default todoSlice.reducer
```

> **Note:** The example above uses `todoReducer` directly as the store reducer, so the root Redux state shape is the slice state itself. If using multiple feature reducers, use an object such as `{ todos: todoReducer }`, then select with `state.todos`.

```jsx
import { useSelector, useDispatch } from 'react-redux'
import {removeTodo} from '../features/todo/todoSlice'

function Todos() {
	const todos = useSelector(state => state.todos)
	const dispatch = useDispatch()

	return (
		<ul className="list-none">
			{todos.map((todo) => (
			<li key={todo.id} >
				<button onClick={() => dispatch(removeTodo(todo.id))} > X </button>
			</li>
			))}
		</ul>
	)
}
```

### Step 5: `App.jsx`

```jsx
function App() {
	return ( <>
		<AddTodo />
		<Todos />
	</>)
}
```

### Step 6: Provide Store

Wrap the application with Redux `Provider`.

```jsx
import { Provider } from 'react-redux'
import {store} from './app/store'

ReactDOM.createRoot(document.getElementById('root')).render(
	<Provider store={store}>
		<App />
	</Provider>,
)
```
